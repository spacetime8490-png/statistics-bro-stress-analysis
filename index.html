# ==============================================================================
# Student Academic Stress & Wellbeing Statistical Analysis Script (Python)
# ==============================================================================
import pandas as pd
import numpy as np

# 1. Load active student dataset
df = pd.read_csv("stats_pro/data/active_dataset.csv")
print(f"Total Cohort Size: {len(df)} students")

# 2. Key Descriptive Statistics
mean_stress = df['stress_level'].mean()
std_stress = df['stress_level'].std(ddof=1)
median_sleep = df['sleep_hours'].median()
burnout_index = df['burnout_index'].mean()
print(f"Mean Stress: {mean_stress:.2f} ± {std_stress:.2f}")
print(f"Median Sleep: {median_sleep:.2f} hours")
print(f"Cohort Mean Burnout Index: {burnout_index:.1f}%")

# 3. Pearson Correlation (Workload vs Stress)
r = np.corrcoef(df['workload'], df['stress_level'])[0, 1]
print(f"Pearson Correlation r(Workload, Stress) = {r:.3f}")

# 4. Independent Two-Sample Comparison (Sleep Deficit vs Normal Sleep)
group_deficit = df[df['sleep_deficit'] == True]['burnout_index']
group_normal = df[df['sleep_deficit'] == False]['burnout_index']
t_stat = (group_deficit.mean() - group_normal.mean()) / np.sqrt(
    group_deficit.var()/len(group_deficit) + group_normal.var()/len(group_normal)
)
print(f"Welch t-Statistic: {t_stat:.3f} (Significant Burnout Disparity)")

# 5. Ordinary Least Squares (OLS) Regression
slope, intercept = np.polyfit(df['workload'], df['stress_level'], 1)
print(f"OLS Regression Line: Stress = {intercept:.2f} + {slope:.2f} * Workload")

data <- read.csv("stats_pro/data/active_dataset.csv")

# 1. Descriptive Summary
summary(data[, c("workload", "sleep_hours", "stress_level", "burnout_index")])

# 2. Pearson & Spearman Correlation
cor.test(data$workload, data$stress_level, method="pearson")
cor.test(data$workload, data$stress_level, method="spearman")

# 3. Two-Sample Welch t-Test
t.test(burnout_index ~ sleep_deficit, data = data)

# 4. One-Way ANOVA across Academic Departments
anova_model <- aov(stress_level ~ branch_code, data = data)
summary(anova_model)

# 5. Multiple Linear Regression Model
model <- lm(burnout_index ~ workload + sleep_hours + screen_time + time_mgmt_diff, data = data)
summary(model)

Universal Statistical Decision Protocol

  1. Step 1 (Formulate Hypotheses): State the Null Hypothesis $H_0$ (no effect, no correlation, equal means) and Alternative Hypothesis $H_1$ (statistically significant effect).
  2. Step 2 (Select Data Columns): Choose the independent variable(s) $X$ and dependent target metric $Y$ from the survey questions.
  3. Step 3 (Calculate Test Statistic): Compute the sample statistic ($t, F, \chi^2, r, z$) by dividing the observed deviation by the standard error.
  4. Step 4 (Evaluate Significance): At confidence level $\alpha = 0.05$, if $p < 0.05$, reject $H_0$ and declare statistical significance with 95% confidence.
  5. Step 5 (Clinical & Academic Impact): Translate the mathematical verdict into student counseling recommendations (e.g. adjust assignment FAT schedules, recommend sleep hygiene workshops).
